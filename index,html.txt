        function showToast(message, type = 'success') {
            const container = document.getElementById('toast-container');

            const styles = {
                success: 'bg-emerald-700 border-emerald-500 text-white',
                error: 'bg-rose-700 border-rose-500 text-white',
                warning: 'bg-amber-600 border-amber-400 text-white',
                info: 'bg-blue-900 border-sky-500 text-white'
            };
            const icons = {
                success: 'fa-circle-check',
                error: 'fa-circle-xmark',
                warning: 'fa-triangle-exclamation',
                info: 'fa-circle-info'
            };

            const toast = document.createElement('div');
            toast.className = `pointer-events-auto flex items-center gap-2.5 px-4 py-3 rounded-xl border shadow-lg text-xs font-semibold transition-all duration-300 ${styles[type] || styles.info}`;
            toast.innerHTML = `<i class="fa-solid ${icons[type] || icons.info} text-sm"></i><span>${message}</span>`;
            container.appendChild(toast);

            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-x-4');
                setTimeout(() => toast.remove(), 300);
            }, 3200);
        }

        function verifyRoleLogin() {
            const selectedRole = document.getElementById('login-role-select').value;
            const enteredPin = document.getElementById('login-pin-input').value.trim();
            const validPin = (db.pins && db.pins[selectedRole]) ? db.pins[selectedRole] : '1234';

            if (enteredPin !== validPin) {
                showToast('PIN salah! Silakan periksa kembali PIN Anda.', 'error');
                document.getElementById('login-pin-input').value = '';
                document.getElementById('login-pin-input').focus();
                return;
            }

            closeModal();
            changeUserRole(selectedRole);
        }

        function savePiketDuty(e) {
            e.preventDefault();
            if (currentRole === 'KEPSEK') {
                showToast('Mode Kepala Sekolah hanya dapat melihat data (Read-Only).', 'warning');
                return;
            }
            const date = document.getElementById('piket-date').value;
            const utamaId = document.getElementById('piket-utama').value;
            const pendampingId = document.getElementById('piket-pendamping').value;
            const start = document.getElementById('piket-start').value;
            const end = document.getElementById('piket-end').value;
            const notes = document.getElementById('piket-notes').value.trim();

            if (!utamaId) {
                showToast('Pilih petugas piket utama terlebih dahulu!', 'error');
                return;
            }

            // Cegah duplikasi data pada tanggal sama (otomatis update jika sudah ada)
            const isUpdate = !!db.piketDuties[date];
            db.piketDuties[date] = { date, utamaId, pendampingId, start, end, notes };
            activeDate = date;
            document.getElementById('global-date').value = date;
            document.getElementById('lap-filter-date').value = date;

            saveDatabase();
            addLog(isUpdate ? 'Update Data Piket' : 'Simpan Data Piket', `Tanggal ${date}`);
            showToast(isUpdate ? 'Data guru piket berhasil diperbarui!' : 'Data guru piket berhasil disimpan!');
            refreshAllViews();
        }

        function updateTeacherStatus(guruId, status) {
            if (currentRole === 'KEPSEK') return;
            ensureTeacherAttendanceDate(activeDate);
            const nowTime = new Date().toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit' });
            const prev = db.teacherAttendance[activeDate][guruId] || { keterangan: '' };

            if (!status) {
                delete db.teacherAttendance[activeDate][guruId];
            } else {
                db.teacherAttendance[activeDate][guruId] = {
                    status,
                    keterangan: prev.keterangan,
                    waktu: nowTime
                };
            }

            const teacher = db.teachers.find(t => t.id === guruId);
            saveDatabase();
            addLog('Presensi Guru', `${teacher?.nama} -> ${status || 'Dikosongkan'} (${activeDate})`);
            
            // Otomatis sinkronkan penanganan kelas jika guru tidak hadir
            syncAbsentHandling(true);
            refreshAllViews();
        }

        function updateTeacherNote(guruId, note) {
            if (currentRole === 'KEPSEK') return;
            ensureTeacherAttendanceDate(activeDate);
            if (!db.teacherAttendance[activeDate][guruId]) {
                db.teacherAttendance[activeDate][guruId] = { status: 'Hadir', keterangan: note, waktu: '-' };
            } else {
                db.teacherAttendance[activeDate][guruId].keterangan = note;
            }
            saveDatabase();
            syncAbsentHandling(false);
            renderDailyReport();
        }

        function markAllTeachersPresent() {
            if (currentRole === 'KEPSEK') return showToast('Akses ditolak untuk mode Kepala Sekolah.', 'warning');
            ensureTeacherAttendanceDate(activeDate);
            const nowTime = new Date().toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit' });

            refreshAllViews();
        }

        // --- 6. LOGIKA PENANGANAN GURU TIDAK HADIR (FITUR UTAMA) ---
        function syncAbsentHandling(notifyUser = false) {
            const dayName = getDayName(activeDate);
            const dayAttendance = db.teacherAttendance[activeDate] || {};
            if (!db.absentHandling[activeDate]) {
                db.absentHandling[activeDate] = [];
            }

            const existingList = db.absentHandling[activeDate];
            const newList = [];
            let newTicketsCount = 0;

            // Cari semua guru yang statusnya Sakit, Izin, atau Alpha pada tanggal ini
            Object.keys(dayAttendance).forEach(guruId => {
                const att = dayAttendance[guruId];
                if (['Sakit', 'Izin', 'Alpha'].includes(att.status)) {
                    // Cari jadwal pelajaran guru tersebut pada hari ini
                    const teacherSchedules = db.schedules.filter(s => s.hari === dayName && s.guruId === guruId);
                    
                    teacherSchedules.forEach(sch => {
                        // Cek apakah sudah ada record penanganan untuk jadwal ini
                        const existing = existingList.find(item => item.scheduleId === sch.id && item.guruId === guruId);
                        if (existing) {
                            existing.statusGuru = att.status;
                            existing.kelas = sch.kelas;
                            existing.mapel = sch.mapel;
                            existing.jamKe = `${sch.jamKe} (${sch.mulai}-${sch.selesai})`;
                            newList.push(existing);
                        } else {
                            newTicketsCount++;
                            newList.push({
                                id: 'H_' + sch.id + '_' + Date.now() + Math.random().toString(36).substr(2, 4),
                                scheduleId: sch.id,
                                guruId: guruId,
                                statusGuru: att.status,
                                kelas: sch.kelas,
                                mapel: sch.mapel,
                                jamKe: `${sch.jamKe} (${sch.mulai}-${sch.selesai})`,
                                tugas: '',
                                penggantiId: '',
                                statusPelaksanaan: 'Belum ditangani',
                                catatan: ''
                            });
                        }
                    });
                }
            });

            db.absentHandling[activeDate] = newList;
            saveDatabase();

            if (notifyUser && newTicketsCount > 0) {
                showToast(`Terdeteksi ${newTicketsCount} jadwal kelas kosong yang perlu ditangani!`, 'warning');
            }
        }

        function saveSchedule(editId) {
            const data = {
                id: editId || 'J_' + Date.now(),
                hari: document.getElementById('sch-hari').value,
                kelas: document.getElementById('sch-kelas').value,
                jamKe: document.getElementById('sch-jam').value.trim(),
                mulai: document.getElementById('sch-mulai').value,
                selesai: document.getElementById('sch-selesai').value,
                guruId: document.getElementById('sch-guru').value,
                mapel: document.getElementById('sch-mapel').value.trim()
            };
            if (!data.jamKe || !data.mapel) return showToast('Lengkapi semua kolom jadwal!', 'error');

            if (editId) {
                const idx = db.schedules.findIndex(s => s.id === editId);
                if (idx !== -1) db.schedules[idx] = data;
            } else {
                db.schedules.push(data);
            }
            saveDatabase();
            addLog(editId ? 'Edit Jadwal' : 'Tambah Jadwal', `${data.hari} Kelas ${data.kelas} (${data.mapel})`);
            closeModal();
            showToast('Jadwal pelajaran berhasil disimpan!');
            syncAbsentHandling(false);
            refreshAllViews();
        }

        function handleImportStudentsExcel(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const data = new Uint8Array(e.target.result);
                    const workbook = XLSX.read(data, { type: 'array' });
                    const firstSheet = workbook.Sheets[workbook.SheetNames[0]];
                    const rows = XLSX.utils.sheet_to_json(firstSheet);

                    let importedCount = 0;
                    rows.forEach((row, idx) => {
                        const nama = row['Nama Siswa'] || row['Nama'] || row['nama'];
                        const nisn = String(row['NISN'] || row['NIS'] || row['nisn'] || '');
                        const kelas = String(row['Kelas'] || row['kelas'] || '').trim().toUpperCase();

                        if (nama && kelas) {
                            if (!db.classes.includes(kelas)) {
                                db.classes.push(kelas);
                            }
                            // Hindari duplikasi NISN
                            const exists = nisn && db.students.some(s => s.nisn === nisn);
                            if (!exists) {
                                db.students.push({
                                    id: 'S_IMP_' + Date.now() + '_' + idx,
                                    nama: String(nama).trim(),
                                    nisn,
                                    kelas
                                });
                                importedCount++;
                            }
                        }
                    });

                    saveDatabase();
                    populateClassSelects();
                    addLog('Impor Excel Siswa', `Berhasil mengimpor ${importedCount} data siswa baru`);
                    showToast(`Berhasil mengimpor ${importedCount} data siswa dari Excel!`);
                    refreshAllViews();
                } catch (err) {
                    console.error(err);
                    showToast('Format file Excel tidak sesuai template!', 'error');
                }
                event.target.value = '';
            };
            reader.readAsArrayBuffer(file);
        }

        // --- 9. DASHBOARD UTAMA & STATISTIK ---
        function computeDailyStats(dateStr) {
            const totalTeachers = db.teachers.length;
            const tAtt = db.teacherAttendance[dateStr] || {};
            let gHadir = 0, gSakit = 0, gIzin = 0, gAlpha = 0;

            db.teachers.forEach(t => {
                const st = tAtt[t.id]?.status;
                if (st === 'Hadir') gHadir++;
                else if (st === 'Sakit') gSakit++;
                else if (st === 'Izin') gIzin++;
                else if (st === 'Alpha') gAlpha++;
            });
            const gAbsen = gSakit + gIzin + gAlpha;

            // Siswa
            const sAttDate = db.studentAttendance[dateStr] || {};
            let sHadir = 0, sSakit = 0, sIzin = 0, sAlpha = 0;
            db.students.forEach(st => {
                const rec = (sAttDate[st.kelas] || {})[st.id];
                const status = rec ? rec.status : 'Hadir'; // Default hadir jika kelas aktif
                if (status === 'Hadir') sHadir++;
                else if (status === 'Sakit') sSakit++;
                else if (status === 'Izin') sIzin++;
                else if (status === 'Alpha') sAlpha++;
            });
            const sAbsen = sSakit + sIzin + sAlpha;

            // Kelas Kosong
            const handlingList = db.absentHandling[dateStr] || [];
            const unhandledCount = handlingList.filter(i => i.statusPelaksanaan === 'Belum ditangani').length;

            return {
                totalTeachers, gHadir, gAbsen, gSakit, gIzin, gAlpha,
                totalStudents: db.students.length, sHadir, sAbsen, sSakit, sIzin, sAlpha,
                totalClassesAffected: handlingList.length,
                unhandledCount,
                handlingList
            };
        }

        function renderDailyReport() {
            const repDate = document.getElementById('lap-filter-date').value || activeDate;
            const searchGuru = (document.getElementById('lap-filter-guru')?.value || '').toLowerCase();
            const stats = computeDailyStats(repDate);
            const duty = db.piketDuties[repDate];

            const formattedDate = formatIndoDate(repDate);
            document.getElementById('rep-subtitle-date').innerText = `Hari/Tanggal: ${formattedDate}`;
            document.getElementById('rep-hari-tanggal').innerText = formattedDate;
            document.getElementById('rep-sign-date').innerText = `Wates, ${formatIndoDateNoDay(repDate)}`;

            const utamaName = duty ? (db.teachers.find(t => t.id === duty.utamaId)?.nama || '-') : '-';
            const pendampingName = duty ? (db.teachers.find(t => t.id === duty.pendampingId)?.nama || '-') : '-';
            document.getElementById('rep-petugas-utama').innerText = utamaName;
            document.getElementById('rep-petugas-pendamping').innerText = pendampingName;
            document.getElementById('rep-jam-tugas').innerText = duty ? `${duty.start} - ${duty.end} WIB` : '-';
            document.getElementById('rep-sign-piket').innerText = utamaName;

            // B. Rekap Guru
            document.getElementById('rep-guru-total').innerText = stats.totalTeachers;
            document.getElementById('rep-guru-h').innerText = stats.gHadir;
            document.getElementById('rep-guru-s').innerText = stats.gSakit;
            document.getElementById('rep-guru-i').innerText = stats.gIzin;
            document.getElementById('rep-guru-a').innerText = stats.gAlpha;

            // C & D. Rekap Guru Tidak Hadir & Tindak Lanjut
            const absenTbody = document.getElementById('rep-table-absen-guru');
            const tAtt = db.teacherAttendance[repDate] || {};
            const absentTeachers = db.teachers.filter(t => {
                const st = tAtt[t.id]?.status;
                const matchStatus = ['Sakit', 'Izin', 'Alpha'].includes(st);
                const matchSearch = !searchGuru || t.nama.toLowerCase().includes(searchGuru);
                return matchStatus && matchSearch;
            });

            if (absentTeachers.length === 0) {
                absenTbody.innerHTML = `<tr><td colspan="6" class="border border-slate-300 p-3 text-center text-slate-500">Nihil (Seluruh guru hadir / sesuai filter pencarian).</td></tr>`;
            } else {
                let rowNum = 1;
                let rowsHtml = '';
                absentTeachers.forEach(teacher => {
                    const att = tAtt[teacher.id];
                    const tickets = stats.handlingList.filter(h => h.guruId === teacher.id);
                    if (tickets.length === 0) {
                        rowsHtml += `
                            <tr>
                                <td class="border border-slate-300 p-2 text-center">${rowNum++}</td>
                                <td class="border border-slate-300 p-2 font-semibold">${teacher.nama}</td>
                                <td class="border border-slate-300 p-2"><b>${att.status}</b> (${att.keterangan || '-'})</td>
                                <td class="border border-slate-300 p-2 text-slate-500">Tidak ada jam mengajar</td>
                                <td class="border border-slate-300 p-2">-</td>
                                <td class="border border-slate-300 p-2">-</td>
                            </tr>
                        `;
                    } else {
                        tickets.forEach(tk => {
                            const pengganti = db.teachers.find(t => t.id === tk.penggantiId)?.nama || 'Petugas Piket';
                            rowsHtml += `
                                <tr>
                                    <td class="border border-slate-300 p-2 text-center">${rowNum++}</td>
                                    <td class="border border-slate-300 p-2 font-semibold">${teacher.nama}</td>
                                    <td class="border border-slate-300 p-2"><b>${att.status}</b> (${att.keterangan || '-'})</td>
                                    <td class="border border-slate-300 p-2">Kelas <b>${tk.kelas}</b> • Jam ${tk.jamKe}<br><i>${tk.mapel}</i></td>
                                    <td class="border border-slate-300 p-2">${tk.tugas || '-'}</td>
                                    <td class="border border-slate-300 p-2"><b>${pengganti}</b><br><span class="text-[11px]">[${tk.statusPelaksanaan}]</span></td>
                                </tr>
                            `;
                        });
                    }
                });
                absenTbody.innerHTML = rowsHtml;
            }

            // E. Rekap Kehadiran Siswa Per Kelas
            const siswaTbody = document.getElementById('rep-table-siswa');
            const sAttDate = db.studentAttendance[repDate] || {};
            siswaTbody.innerHTML = db.classes.map(cls => {
                const classStudents = db.students.filter(s => s.kelas === cls);
                const total = classStudents.length;
                let h = 0, s = 0, i = 0, a = 0;
                classStudents.forEach(st => {
                    const status = (sAttDate[cls] || {})[st.id]?.status || 'Hadir';
                    if (status === 'Hadir') h++;
                    else if (status === 'Sakit') s++;
                    else if (status === 'Izin') i++;
                    else if (status === 'Alpha') a++;
                });
                const pct = total > 0 ? ((h / total) * 100).toFixed(1) + '%' : '-';
                return `
                    <tr>
                        <td class="border border-slate-300 p-1.5 font-bold">${cls}</td>
                        <td class="border border-slate-300 p-1.5">${total}</td>
                        <td class="border border-slate-300 p-1.5">${h}</td>
                        <td class="border border-slate-300 p-1.5">${s}</td>
                        <td class="border border-slate-300 p-1.5">${i}</td>
                        <td class="border border-slate-300 p-1.5">${a}</td>
                        <td class="border border-slate-300 p-1.5 font-semibold">${pct}</td>
                    </tr>
                `;
            }).join('');

            // F & G. Catatan & Pengesahan
            document.getElementById('rep-catatan-piket').innerText = duty?.notes || 'Situasi kegiatan belajar mengajar berjalan aman, tertib, dan kondusif.';
            document.getElementById('rep-sign-kepsek').innerText = db.kepsek.nama;
            document.getElementById('rep-nip-kepsek').innerText = `NIP. ${db.kepsek.nip}`;
        }

        function renderMonitoring() {
            // Hitung agregasi seluruh data tersimpan
            let totalTeacherSlots = 0, totalTeacherPresent = 0;
            const teacherAbsenMap = {};
            db.teachers.forEach(t => teacherAbsenMap[t.id] = { nama: t.nama, s: 0, i: 0, a: 0 });

            Object.keys(db.teacherAttendance).forEach(d => {
                const rec = db.teacherAttendance[d];
                Object.keys(rec).forEach(gid => {
                    const st = rec[gid].status;
                    if (st) {
                        totalTeacherSlots++;
                        if (st === 'Hadir') totalTeacherPresent++;
                        else if (teacherAbsenMap[gid]) {
                            if (st === 'Sakit') teacherAbsenMap[gid].s++;
                            if (st === 'Izin') teacherAbsenMap[gid].i++;
                            if (st === 'Alpha') teacherAbsenMap[gid].a++;
                        }
                    }
                });
            });

            const pctGuru = totalTeacherSlots > 0 ? ((totalTeacherPresent / totalTeacherSlots) * 100).toFixed(1) + '%' : '100%';
            document.getElementById('mon-kpi-guru').innerText = pctGuru;

            // Penyelesaian tugas kelas kosong & kelas sering ditinggalkan
            let totalTickets = 0, solvedTickets = 0;
            const classEmptyCount = {};
            Object.keys(db.absentHandling).forEach(d => {
                db.absentHandling[d].forEach(tk => {
                    totalTickets++;
                    if (tk.statusPelaksanaan !== 'Belum ditangani') solvedTickets++;
                    classEmptyCount[tk.kelas] = (classEmptyCount[tk.kelas] || 0) + 1;
                });
            });
            const pctTugas = totalTickets > 0 ? ((solvedTickets / totalTickets) * 100).toFixed(0) + '%' : '100%';
            document.getElementById('mon-kpi-tugas').innerText = pctTugas;

            // Tabel guru sering tidak hadir
            const absenList = Object.values(teacherAbsenMap)
                .map(x => ({ ...x, total: x.s + x.i + x.a }))
                .filter(x => x.total > 0)
                .sort((a, b) => b.total - a.total);

            document.getElementById('mon-table-guru-absen').innerHTML = absenList.length === 0
                ? `<tr><td colspan="5" class="p-4 text-center text-slate-400">Belum ada catatan ketidakhadiran guru.</td></tr>`
                : absenList.map(g => `
                    <tr>
                        <td class="p-2.5 font-semibold text-slate-800">${g.nama}</td>
                        <td class="p-2.5 text-center">${g.s}</td>
                        <td class="p-2.5 text-center">${g.i}</td>
                        <td class="p-2.5 text-center">${g.a}</td>
                        <td class="p-2.5 text-center font-bold text-rose-600">${g.total}</td>
                    </tr>
                `).join('');

            // Tabel kelas sering ditinggalkan
            const classList = Object.entries(classEmptyCount).sort((a, b) => b[1] - a[1]);
            document.getElementById('mon-table-kelas-kosong').innerHTML = classList.length === 0
                ? `<tr><td colspan="3" class="p-4 text-center text-slate-400">Belum ada kelas yang ditinggalkan guru.</td></tr>`
                : classList.map(([cls, cnt]) => `
                    <tr>
                        <td class="p-2.5 font-bold text-blue-950">Kelas ${cls}</td>
                        <td class="p-2.5 text-center font-semibold">${cnt} Sesi</td>
                        <td class="p-2.5 text-center"><span class="px-2 py-0.5 rounded-full bg-emerald-100 text-emerald-700 font-semibold">Terpantau Piket</span></td>
                    </tr>
                `).join('');

            // Render Chart Periode
            const chartConfigs = {
                'mingguan': {
                    title: 'Tren Kehadiran Guru & Siswa (Periode Mingguan)',
                    labels: ['Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu'],
                    guruData: [96, 98, 95, 97, 99, 98],
                    siswaData: [98, 97, 98, 96, 99, 97]
                },
                'bulanan': {
                    title: 'Tren Kehadiran Guru & Siswa (Periode Bulanan)',
                    labels: ['Minggu 1', 'Minggu 2', 'Minggu 3', 'Minggu 4'],
                    guruData: [97.2, 96.5, 98.0, 97.8],
                    siswaData: [98.1, 97.9, 98.4, 98.2]
                },
                'semester': {
                    title: 'Tren Kehadiran Guru & Siswa (Semester Ganjil TP 2026/2027)',
                    labels: ['Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember'],
                    guruData: [98.5, 97.0, 96.8, 97.5, 97.2, 98.0],
                    siswaData: [98.8, 98.2, 97.9, 98.1, 98.3, 98.5]
                }
            };

            const cfg = chartConfigs[monPeriod];
            document.getElementById('mon-chart-title').innerText = cfg.title;
            const ctx = document.getElementById('monitoringChart');
            if (monChartInstance) monChartInstance.destroy();

            monChartInstance = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: cfg.labels,
                    datasets: [
                        {
                            label: '% Kehadiran Guru',
                            data: cfg.guruData,
                            borderColor: '#1e3a8a',
                            backgroundColor: 'rgba(30, 58, 138, 0.1)',
                            fill: true,
                            tension: 0.35
                        },
                        {
                            label: '% Kehadiran Siswa',
                            data: cfg.siswaData,
                            borderColor: '#0ea5e9',
                            backgroundColor: 'rgba(14, 165, 233, 0.1)',
                            fill: true,
                            tension: 0.35
                        }
                    ]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: { y: { min: 85, max: 100 } }
                }
            });

        function saveTeacher(editId) {
            const nama = document.getElementById('m-guru-nama').value.trim();
            const nip = document.getElementById('m-guru-nip').value.trim();
            const mapel = document.getElementById('m-guru-mapel').value.trim();
            if (!nama || !mapel) return showToast('Nama dan mata pelajaran wajib diisi!', 'error');

            if (editId) {
                const idx = db.teachers.findIndex(x => x.id === editId);
                if (idx !== -1) db.teachers[idx] = { id: editId, nama, nip, mapel };
            } else {
                db.teachers.push({ id: 'G_' + Date.now(), nama, nip, mapel });
            }
            saveDatabase();
            addLog(editId ? 'Edit Data Guru' : 'Tambah Guru', nama);
            closeModal();
            populateTeacherSelects();
            showToast('Database guru berhasil diperbarui!');
            refreshAllViews();
        }

        function deleteTeacher(id) {
            const t = db.teachers.find(x => x.id === id);
            db.teachers = db.teachers.filter(x => x.id !== id);
            saveDatabase();
            addLog('Hapus Data Guru', t?.nama || id);
            populateTeacherSelects();
            showToast('Data guru dihapus.', 'info');
            refreshAllViews();
        }

        function saveKepsekSettings() {

        function savePinSettings() {
            if (currentRole !== 'ADMIN') return;
